Pack Physical value → 1×8 uint8; [] creates a frame

function p = can_pack(p,s,phys)
if isempty(p), p = zeros(1,8,'uint8'); end
m = bitshift(uint64(1),s.length)-1;    % mask: 2^len-1
r = round((phys-s.offset)/s.scale);   % physical -> raw
if s.signed&&r<0, r=r+2^s.length; end % two's complement
w = typecast(p,'uint64');
p = typecast(bitor(bitand(w,bitcmp(bitshift(m,s.startBit))),...
                  bitshift(bitand(uint64(r),m),s.startBit)),'uint8');
end

Unpack 1×8 uint8 → physical value

function phys = can_unpack(s,p)
m = bitshift(uint64(1),s.length)-1;
r = double(bitand(bitshift(typecast(p,'uint64'),-s.startBit),m));
if s.signed&&r>=2^(s.length-1), r=r-2^s.length; end % signed raw
phys = r*s.scale + s.offset;
end

Usage Pack all signals / unpack one signal

sigs = can_protocol_def();
phys = [398.6 -125.7 76.4 34 3.652 1 3 5];
p = []; for k=1:8, p = can_pack(p,sigs(k),phys(k)); end
fprintf('%02X ',p)                    % 92 7F B1 BF 22 44 0E 57
can_unpack(sigs(2),p)                 % -125.7
find(can_pack(p,sigs(6),0)~=p)-1      % 7 (Byte7 only)
p(4) = uint8(191);                    % aligned byte: SOC = 76.4%
  • Byte0 = payload(1); sent first. startBit = LSB position.
  • Cross-byte: bit index +1; bit7 → next byte's bit0.
  • 16-bit Intel: low byte first (44 0E), not 0E 44.
  • Negative raw: add 2^len, then cast and mask.
  • Quantization: always round the raw value.
  • Aligned 8-bit (unsigned): p(b)=uint8(raw), b=startBit/8+1.
